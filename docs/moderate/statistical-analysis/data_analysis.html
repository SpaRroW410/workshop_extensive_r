Grouped summaries

summary_table <- p %>%
  mutate(wealth_index = factor(wealth_index_of_participants)) %>% 
  group_by(wealth_index) %>%
  reframe(
    Frequency = n(),
    Percentage = paste0(round(n() * 100 / nrow(p), 2), "%")
  )

Hypothesis testing

Chi-square goodness-of-fit

observed <- table(p$ever_screened_for_bp)
expected <- c(1/2, 1/2) * sum(observed)
chisq_test <- chisq.test(x = observed, p = expected / sum(observed))
chisq_test

    Chi-squared test for given probabilities

data:  observed
X-squared = 80.23, df = 1, p-value < 2.2e-16

Chi-square test for association

table_gender_dm <- table(p$sex_of_participants, p$past_history_of_diabetes)
chisq_m_5_b <- chisq.test(table_gender_dm, correct = FALSE)
chisq_m_5_b

    Pearson's Chi-squared test

data:  table_gender_dm
X-squared = 0.40978, df = 1, p-value = 0.5221

McNemar test for paired binary responses

table_bfr_aftr <- table(p$pain_foot_bfr, p$pain_foot_aftr)
mcnemar_test <- mcnemar.test(table_bfr_aftr)
mcnemar_test

    McNemar's Chi-squared test with continuity correction

data:  table_bfr_aftr
McNemar's chi-squared = 0.20712, df = 1, p-value = 0.649
