# Two-sample t-test
t.test(blood_pressure ~ treatment, data = data)

Chi-square test

# Create categorical data
data$outcome <- sample(c("Improved", "Not Improved"), 100, replace = TRUE)

# Chi-square test
chisq.test(table(data$treatment, data$outcome))

    Pearson's Chi-squared test with Yates' continuity correction
