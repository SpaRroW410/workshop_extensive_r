Using gtsummary for Professional Tables

gtsummary creates publication-ready tables from R data.

Basic Summary Table

library(gtsummary)

data <- gtsummary::trial
# Summary table

  tbl_summary(by = trt) %>%

Regression Tables

# Logistic regression example
data$outcome <- rbinom(200, 1, 0.5)

model_log <- glm(outcome ~ age + trt, data = data, family = binomial)

tbl_regression(model_log, exponentiate = TRUE)

Customizing Tables

data %>% 
  tbl_summary(
    by = trt,
    statistic = list(
      all_continuous() ~ "{mean} ({sd})",
      all_categorical() ~ "{n} ({p}%)"
    )
  ) %>% 
  add_p() %>% 
  modify_header(label = "**Variable**") %>% 
  bold_labels()
